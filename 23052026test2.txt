-- 1. List the Employee Name, Department Name, and Location Name for all employees.
SELECT e.employee_name,
       d.dept_name,
       d.location_id
FROM Employees e
JOIN Departments d
ON e.dept_id = d.dept_id;


-- 2. Retrieve the Project Name, Client Name, and Total Hours Worked
-- for projects managed by the 'Consulting' department.
SELECT p.project_name,
       p.client_name,
       SUM(a.hours_worked) AS total_hours
FROM Projects p
JOIN Assignments a
ON p.project_id = a.project_id
JOIN Departments d
ON p.dept_id = d.dept_id
WHERE d.dept_name = 'Consulting'
GROUP BY p.project_name, p.client_name;


-- 3. Find employees who worked on projects for 'GlobalTech Solutions'.
SELECT e.employee_name,
       p.project_name
FROM Employees e
JOIN Assignments a
ON e.employee_id = a.employee_id
JOIN Projects p
ON a.project_id = p.project_id
WHERE p.client_name = 'GlobalTech Solutions';


-- 4. List all departments that have no employees assigned.
SELECT d.dept_name
FROM Departments d
LEFT JOIN Employees e
ON d.dept_id = e.dept_id
WHERE e.employee_id IS NULL;


-- 5. Show manager name, department name, and total direct reports.
SELECT m.employee_name AS manager_name,
       d.dept_name,
       COUNT(e.employee_id) AS total_reports
FROM Employees e
JOIN Employees m
ON e.manager_id = m.employee_id
JOIN Departments d
ON m.dept_id = d.dept_id
GROUP BY m.employee_name, d.dept_name;


-- 6. Calculate average hours worked per employee across all projects.
SELECT e.employee_name,
       AVG(a.hours_worked) AS avg_hours
FROM Employees e
JOIN Assignments a
ON e.employee_id = a.employee_id
GROUP BY e.employee_name;


-- 7. Find departments with average salary above company-wide average salary.
SELECT d.dept_name,
       AVG(e.salary) AS avg_salary
FROM Employees e
JOIN Departments d
ON e.dept_id = d.dept_id
GROUP BY d.dept_name
HAVING AVG(e.salary) >
(
    SELECT AVG(salary)
    FROM Employees
);


-- 8. Identify clients whose projects exceed 500 total hours worked.
SELECT p.client_name,
       SUM(a.hours_worked) AS total_hours
FROM Projects p
JOIN Assignments a
ON p.project_id = a.project_id
GROUP BY p.client_name
HAVING SUM(a.hours_worked) > 500;


-- 9. For each location, find the department with highest employee count.
SELECT location_id,
       dept_name,
       employee_count
FROM
(
    SELECT d.location_id,
           d.dept_name,
           COUNT(e.employee_id) AS employee_count,
           RANK() OVER
           (PARTITION BY d.location_id
            ORDER BY COUNT(e.employee_id) DESC) AS rnk
    FROM Departments d
    LEFT JOIN Employees e
    ON d.dept_id = e.dept_id
    GROUP BY d.location_id, d.dept_name
) x
WHERE rnk = 1;


-- 10. Total employees assigned to each project
-- only include projects with more than 5 employees assigned.
SELECT p.project_name,
       COUNT(DISTINCT a.employee_id) AS total_employees
FROM Projects p
JOIN Assignments a
ON p.project_id = a.project_id
GROUP BY p.project_name
HAVING COUNT(DISTINCT a.employee_id) > 5;


-- 11. Find employees who work on ALL projects managed by the 'R&D' department.
SELECT e.employee_name
FROM Employees e
WHERE NOT EXISTS
(
    SELECT p.project_id
    FROM Projects p
    JOIN Departments d
    ON p.dept_id = d.dept_id
    WHERE d.dept_name = 'R&D'
    AND NOT EXISTS
    (
        SELECT 1
        FROM Assignments a
        WHERE a.employee_id = e.employee_id
        AND a.project_id = p.project_id
    )
);


-- 12. Retrieve departments with employees assigned to more than one project.
SELECT DISTINCT d.dept_name
FROM Departments d
JOIN Employees e
ON d.dept_id = e.dept_id
JOIN Assignments a
ON e.employee_id = a.employee_id
GROUP BY d.dept_name, e.employee_id
HAVING COUNT(DISTINCT a.project_id) > 1;


-- 13. List employee name and project name where hours worked < 10.
SELECT e.employee_name,
       p.project_name
FROM Employees e
JOIN Assignments a
ON e.employee_id = a.employee_id
JOIN Projects p
ON a.project_id = p.project_id
WHERE a.hours_worked < 10;


-- 14. Find projects with no assigned employees.
SELECT p.project_name
FROM Projects p
LEFT JOIN Assignments a
ON p.project_id = a.project_id
WHERE a.assignment_id IS NULL;


-- 15. Employees whose salary is greater than maximum salary in their department.
-- (No employee will satisfy logically, but query is:)
SELECT employee_name
FROM Employees e1
WHERE salary >
(
    SELECT MAX(e2.salary)
    FROM Employees e2
    WHERE e1.dept_id = e2.dept_id
);


-- 16. Find employee name and oldest project start date.
SELECT e.employee_name,
       MIN(a.start_date) AS oldest_project_date
FROM Employees e
JOIN Assignments a
ON e.employee_id = a.employee_id
GROUP BY e.employee_name;


-- 17. List all projects active in year 2023 using DATE_FORMAT.
SELECT project_name,
       start_date
FROM Projects
WHERE DATE_FORMAT(start_date, '%Y') = '2023';


-- 18. Use IFNULL to display CEO instead of NULL manager.
SELECT employee_name,
       IFNULL(CAST(manager_id AS CHAR), 'CEO') AS manager
FROM Employees;
